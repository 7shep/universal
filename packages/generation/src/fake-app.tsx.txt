import { useEffect, useState } from 'react';

const knownRoutes = ['/', '/keyboards/monolith-75', '/craft'] as const;
type Route = (typeof knownRoutes)[number];
const currentRoute = (): Route =>
  knownRoutes.find((route) => route === window.location.pathname) ?? '/';
function go(path: Route) {
  window.history.pushState({}, '', path);
  window.dispatchEvent(new PopStateEvent('popstate'));
}
function Link({ to, children }: { to: Route; children: React.ReactNode }) {
  return (
    <a
      href={to}
      onClick={(event) => {
        event.preventDefault();
        go(to);
      }}
    >
      {children}
    </a>
  );
}

        className="route-mark"
        aria-label={`Current page: ${route === '/' ? 'Home' : route.includes('keyboards') ? 'Product' : 'Craft'}`}

        {route === '/' ? '01' : route.includes('keyboards') ? '02' : '03'} / 03

}

export default function App() {
  const [route, setRoute] = useState<Route>(currentRoute);
  useEffect(() => {
    const update = () => setRoute(currentRoute());
    window.addEventListener('popstate', update);
    return () => window.removeEventListener('popstate', update);
  }, []);
  return (
    <>
      <a className="skip" href="#main">
        Skip to content
      </a>
      <Navigation route={route} />
      {route === '/' ? <Home /> : route.includes('keyboards') ? <Product /> : <Craft />}
